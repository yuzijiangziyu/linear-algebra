\chapter{Vector Spaces}

\section{Fields and vector spaces}

We begin with the scalars. A \emph{field} is a set $F$ equipped with addition and multiplication such that $F$ is an Abelian group under addition with identity $0$, $F^\times\equiv F\backslash\{0\}$ is an Abelian group under multiplication with identity $1$, and multiplication distributes over addition. We always assume $0\neq 1$. Standard examples are $\Q$, $\R$, $\mathbb{C}$, and finite fields such as $\Fp$.

\begin{definition}
	A \emph{vector space} over a field $F$ is an Abelian group $(V,+)$ together with a scalar multiplication
	\[
	F\times V\longrightarrow V,\qquad (a,v)\longmapsto av,
	\]
	such that for all $a,b\in F$ and $u,v\in V$,
	\[
	\begin{split}
	a(u+v)&=au+av,\\
	(a+b)v&=av+bv,\\
	(ab)v&=a(bv),\\
	1v&=v.
	\end{split}
	\]
\end{definition}

The additive identity of $V$ will also be denoted by $0$. The usual consequences, such as $0v=0$, $a0=0$, and $(-1)v=-v$, follow directly from the axioms.

The most familiar example is $F^n$ with componentwise addition and scalar multiplication. Other examples include the polynomial space $F[t]$ and, for any set $X$, the space of all functions $X\to F$ under pointwise operations. These examples already show that a vector space need not be finite-dimensional.

When the base field is understood, we will simply call $V$ a vector space.

\section{Linear combinations and dependence}

The first notions we need require only the vector-space axioms.

\begin{definition}
	Let $v_1,\dots,v_n\in V$. A \emph{linear combination} of $v_1,\dots,v_n$ is a vector of the form
	\[
	a_1v_1+\cdots+a_nv_n
	\]
	for scalars $a_1,\dots,a_n\in F$. By convention, the empty linear combination is $0$.
\end{definition}

\begin{definition}
	A subset $S\subseteq V$ is \emph{linearly independent} if every relation
	\[
	a_1v_1+\cdots+a_nv_n=0,
	\qquad v_1,\dots,v_n\in S \text{ distinct},
	\]
	implies $a_1=\cdots=a_n=0$. Otherwise $S$ is \emph{linearly dependent}. The empty set is linearly independent.
\end{definition}

Thus linear independence is a condition on finite linear relations even when $S$ itself is not assumed finite. A subset of a linearly independent set is linearly independent, while any superset of a linearly dependent set is linearly dependent.

The following theorem is the basic counting result from which the dimension theory of finite-dimensional vector spaces will follow.

\begin{theorem}[Fundamental theorem on linear dependence]\label{thm:fundamental-dependence}
	Let $x_1,\dots,x_n\in V$. If $y_1,\dots,y_{n+1}$ are each linear combinations of $x_1,\dots,x_n$, then $y_1,\dots,y_{n+1}$ are linearly dependent.
\end{theorem}

\begin{proof}
	We argue by induction on $n$. If $n=0$, every $y_j$ is the empty linear combination and hence is $0$, so the conclusion is immediate.

	Suppose $n>0$ and the result is known for $n-1$. Write
	\[
	y_j=z_j+a_jx_n,
	\]
	where each $z_j$ is a linear combination of $x_1,\dots,x_{n-1}$.

	If every $a_j$ is zero, then already $y_1,\dots,y_n$ are $n$ vectors which are linear combinations of $n-1$ vectors, so they are linearly dependent by the induction hypothesis.

	Otherwise, after relabeling, assume $a_{n+1}\neq0$. For $1\le i\le n$, set
	\[
	y_i'=y_i-\frac{a_i}{a_{n+1}}y_{n+1}.
	\]
	Each $y_i'$ is a linear combination of $x_1,\dots,x_{n-1}$. By the induction hypothesis, there are scalars $b_1,\dots,b_n$, not all zero, such that
	\[
	\sum_{i=1}^n b_i y_i'=0.
	\]
	Substituting the definition of $y_i'$ gives
	\[
	\sum_{i=1}^n b_i y_i
	-\left(\sum_{i=1}^n b_i\frac{a_i}{a_{n+1}}\right)y_{n+1}=0.
	\]
	Since the coefficients $b_1,\dots,b_n$ are not all zero, this is a nontrivial linear relation among $y_1,\dots,y_{n+1}$.
\end{proof}

\begin{corollary}\label{cor:independent-vs-generators}
	Suppose $y_1,\dots,y_m$ are linearly independent and each $y_j$ is a linear combination of $x_1,\dots,x_n$. Then $m\le n$.
\end{corollary}

\begin{proof}
	If $m>n$, any $n+1$ of the $y_j$ would be linearly dependent by Theorem \ref{thm:fundamental-dependence}.
\end{proof}

This simple inequality is the only counting result we will need to construct dimension.

\section{Subspaces and span}\label{sec:subspace}

\begin{definition}
	A subset $W\subseteq V$ is a \emph{subspace} of $V$ if $W$ is itself a vector space under the operations inherited from $V$.
\end{definition}

For practical purposes, a nonempty subset $W\subseteq V$ is a subspace if and only if
\[
au+bv\in W
\]
for all $u,v\in W$ and $a,b\in F$.

\begin{definition}
	For a subset $S\subseteq V$, the \emph{span} of $S$, denoted by $\operatorname{span}(S)$, is the set of all finite linear combinations of elements of $S$. By convention,
	\[
	\operatorname{span}(\emptyset)=\{0\}.
	\]
	We say that $S$ \emph{spans} or \emph{generates} $V$ if $\operatorname{span}(S)=V$.
\end{definition}

\begin{proposition}\label{prop:span-smallest}
	$\operatorname{span}(S)$ is the smallest subspace of $V$ containing $S$.
\end{proposition}

\begin{proof}
	A linear combination of linear combinations of elements of $S$ is again a linear combination of elements of $S$, so $\operatorname{span}(S)$ is a subspace. If $W$ is any subspace containing $S$, then $W$ contains every finite linear combination of elements of $S$. Hence
	\[
	\operatorname{span}(S)\subseteq W.
	\]
\end{proof}

Equivalently,
\[
\operatorname{span}(S)
=
\bigcap_{\substack{W\le V\\S\subseteq W}}W.
\]

The following elementary observation will be used repeatedly.

\begin{lemma}\label{lem:add-independent}
	Let $S$ be linearly independent and let $v\in V$. Then $S\cup\{v\}$ is linearly independent if and only if $v\notin\operatorname{span}(S)$.
\end{lemma}

\begin{proof}
	If $v\in\operatorname{span}(S)$, moving a linear expression for $v$ to one side produces a nontrivial relation on $S\cup\{v\}$. Conversely, if $S\cup\{v\}$ is dependent, a nontrivial relation cannot involve only elements of $S$. Therefore the coefficient of $v$ is nonzero, and solving for $v$ expresses it as a linear combination of elements of $S$.
\end{proof}

\section{Bases and dimension}

\begin{definition}
	A \emph{basis} of $V$ is a linearly independent subset of $V$ which spans $V$.
\end{definition}

A basis is intrinsically an unordered set. An \emph{ordered basis} $(v_1,\dots,v_n)$ is a basis together with a choice of order; the order becomes relevant when coordinates and matrices are introduced.

\begin{proposition}\label{prop:finite-spanning-basis}
	Every finite spanning set contains a basis.
\end{proposition}

\begin{proof}
	Let $S$ be a finite spanning set. If $S$ is linearly independent, it is already a basis. Otherwise some nontrivial linear relation among elements of $S$ allows one element of $S$ to be written as a linear combination of the others. Removing that element does not change the span. Repeating this process must terminate, since $S$ is finite, and the resulting spanning set is linearly independent.
\end{proof}

\begin{theorem}\label{thm:basis-cardinality}
	If a vector space has a finite basis, then every basis has the same finite number of elements.
\end{theorem}

\begin{proof}
	Let $B=\{v_1,\dots,v_n\}$ be a basis and let $B'$ be any other basis. Since $B'$ is linearly independent and every element of $B'$ is a linear combination of $B$, Corollary \ref{cor:independent-vs-generators} shows that $B'$ contains at most $n$ elements. In particular $B'$ is finite.

	If $|B'|=m$, then the same argument with $B$ and $B'$ interchanged gives $n\le m$. Hence $m=n$.
\end{proof}

\begin{definition}
	A vector space is \emph{finite-dimensional} if it is spanned by finitely many vectors. Its \emph{dimension}, denoted by $\dim V$, is the number of elements in any basis. This is well-defined by Proposition \ref{prop:finite-spanning-basis} and Theorem \ref{thm:basis-cardinality}. We set $\dim\{0\}=0$.
\end{definition}

Thus, if $\dim V=n$, every basis of $V$ has exactly $n$ elements. The fundamental dependence theorem immediately gives the two basic inequalities:
\[
\boxed{
\text{size of an independent set}
\ \le\ \dim V
\ \le\
\text{size of a finite spanning set}.
}
\]

\begin{theorem}[Basis extension]\label{thm:basis-extension}
	Every linearly independent subset of a finite-dimensional vector space can be extended to a basis.
\end{theorem}

\begin{proof}
	Let $\dim V=n$ and let $S$ be linearly independent. By Corollary \ref{cor:independent-vs-generators}, $S$ contains at most $n$ elements and is therefore finite.

	If $\operatorname{span}(S)=V$, then $S$ is already a basis. Otherwise choose
	\[
	v\in V\backslash\operatorname{span}(S).
	\]
	By Lemma \ref{lem:add-independent}, $S\cup\{v\}$ is still linearly independent. Repeat. The process cannot produce more than $n$ independent vectors, so after finitely many steps it must span $V$.
\end{proof}

\begin{corollary}\label{cor:n-independent-basis}
	If $\dim V=n$, then any $n$ linearly independent vectors form a basis.
\end{corollary}

\begin{corollary}\label{cor:n-spanning-basis}
	If $\dim V=n$, then any spanning set consisting of exactly $n$ vectors is a basis.
\end{corollary}

\begin{proof}
	If such a spanning set were dependent, one of its elements could be removed without changing the span. This would give a spanning set of $n-1$ vectors, contradicting Corollary \ref{cor:independent-vs-generators} applied to any basis of $V$.
\end{proof}

\begin{proposition}[Coordinates]\label{prop:coordinates}
	Let $(v_1,\dots,v_n)$ be an ordered basis of $V$. Every $v\in V$ can be written uniquely as
	\[
	v=a_1v_1+\cdots+a_nv_n.
	\]
\end{proposition}

\begin{proof}
	Existence follows because the basis spans $V$. If
	\[
	a_1v_1+\cdots+a_nv_n
	=
	b_1v_1+\cdots+b_nv_n,
	\]
	then
	\[
	(a_1-b_1)v_1+\cdots+(a_n-b_n)v_n=0.
	\]
	Linear independence gives $a_i=b_i$ for every $i$.
\end{proof}

The tuple $(a_1,\dots,a_n)$ is the coordinate of $v$ with respect to the ordered basis. Coordinates depend on the chosen basis, while dimension does not.

\begin{proposition}\label{prop:subspace-dimension}
	Every subspace $W$ of a finite-dimensional vector space $V$ is finite-dimensional and
	\[
	\dim W\le\dim V.
	\]
	Moreover, equality holds if and only if $W=V$.
\end{proposition}

\begin{proof}
	Starting from the empty set, repeatedly choose a vector of $W$ outside the span of those already chosen. Lemma \ref{lem:add-independent} shows that the chosen vectors remain linearly independent. The process must stop after at most $\dim V$ steps, and when it stops the chosen vectors span $W$. Thus they form a basis of $W$.

	If $\dim W=\dim V=n$, a basis of $W$ consists of $n$ linearly independent vectors in $V$, hence is a basis of $V$ by Corollary \ref{cor:n-independent-basis}. Therefore $W=V$.
\end{proof}

\section{Sums and direct sums}

Let $U$ and $W$ be subspaces of $V$. Their sum is
\[
U+W=\{u+w\mid u\in U,\ w\in W\}.
\]
It is again a subspace of $V$.

\begin{theorem}[Dimension formula]\label{thm:dimension-sum}
	If $U$ and $W$ are finite-dimensional subspaces, then
	\[
	\dim(U+W)
	=
	\dim U+\dim W-\dim(U\cap W).
	\]
\end{theorem}

\begin{proof}
	Choose a basis $c_1,\dots,c_r$ of $U\cap W$. Extend it to bases
	\[
	c_1,\dots,c_r,u_1,\dots,u_p
	\]
	of $U$ and
	\[
	c_1,\dots,c_r,w_1,\dots,w_q
	\]
	of $W$. Then
	\[
	c_1,\dots,c_r,u_1,\dots,u_p,w_1,\dots,w_q
	\]
	is a basis of $U+W$.

	It clearly spans $U+W$. To see independence, suppose a linear combination of these vectors is zero. Move the $u_i$ terms to one side. Their sum then lies both in $U$ and in $W$, hence in $U\cap W$. Independence of the chosen basis of $U$ forces all coefficients of the $u_i$ to vanish. The remaining relation lies in the chosen basis of $W$, so all remaining coefficients also vanish.

	Therefore
	\[
	\dim(U+W)=r+p+q=(r+p)+(r+q)-r.
	\]
\end{proof}

\begin{definition}
	We say that $V$ is the \emph{direct sum} of subspaces $U$ and $W$, and write
	\[
	V=U\oplus W,
	\]
	if $V=U+W$ and $U\cap W=\{0\}$.
\end{definition}

Equivalently, $V=U\oplus W$ if and only if every $v\in V$ has a unique expression
\[
v=u+w,\qquad u\in U,\quad w\in W.
\]

In a finite-dimensional space,
\[
V=U\oplus W
\qquad\Longrightarrow\qquad
\dim V=\dim U+\dim W.
\]

\begin{proposition}[Existence of complements]\label{prop:complement}
	Every subspace $W$ of a finite-dimensional vector space $V$ has a complementary subspace $U$ such that
	\[
	V=W\oplus U.
	\]
\end{proposition}

\begin{proof}
	Choose a basis of $W$ and extend it to a basis of $V$. Let $U$ be the span of the additional basis vectors.
\end{proof}

The complement is generally not unique.

\section{Quotient spaces}

Let $W$ be a subspace of $V$. For $v\in V$, define the coset
\[
v+W=\{v+w\mid w\in W\}.
\]
Two vectors determine the same coset precisely when their difference lies in $W$.

\begin{definition}
	The \emph{quotient space} $V/W$ is the set of all cosets $v+W$, with operations
	\[
	(v+W)+(u+W)=(v+u)+W
	\]
	and
	\[
	a(v+W)=av+W.
	\]
\end{definition}

These operations are well-defined: changing $v$ or $u$ by elements of $W$ changes the resulting representative only by another element of $W$. With these operations, $V/W$ is a vector space over $F$.

\begin{theorem}\label{thm:quotient-dimension}
	If $V$ is finite-dimensional and $W\le V$, then
	\[
	\dim(V/W)=\dim V-\dim W.
	\]
\end{theorem}

\begin{proof}
	Let $w_1,\dots,w_r$ be a basis of $W$, and extend it to a basis
	\[
	w_1,\dots,w_r,v_1,\dots,v_s
	\]
	of $V$. Then
	\[
	v_1+W,\dots,v_s+W
	\]
	is a basis of $V/W$.

	Indeed, every coset has a representative whose $w_i$ components have been discarded, so these cosets span $V/W$. If
	\[
	a_1(v_1+W)+\cdots+a_s(v_s+W)=W,
	\]
	then $a_1v_1+\cdots+a_sv_s\in W$, and independence of the basis of $V$ forces $a_1=\cdots=a_s=0$.
\end{proof}

Thus passing to a quotient removes exactly the dimensions belonging to the subspace being collapsed.

\section{Isomorphism}

We finish the chapter with the intrinsic classification of finite-dimensional vector spaces. Linear maps will be studied systematically in the next chapter.

\begin{definition}
	Two vector spaces $V$ and $W$ over the same field $F$ are \emph{isomorphic} if there exists a bijection
	\[
	\tau\colon V\longrightarrow W,\qquad v\longmapsto v^\tau,
	\]
	such that
	\[
	(au+bv)^\tau=au^\tau+bv^\tau
	\]
	for all $u,v\in V$ and $a,b\in F$. Such a map is called an \emph{isomorphism}.
\end{definition}

An isomorphism carries linearly independent sets to linearly independent sets and spanning sets to spanning sets; in particular, it carries bases to bases.

\begin{theorem}\label{thm:dimension-classification}
	Two finite-dimensional vector spaces over the same field are isomorphic if and only if they have the same dimension.
\end{theorem}

\begin{proof}
	If $V$ and $W$ are isomorphic, an isomorphism sends a basis of $V$ to a basis of $W$, so their dimensions are equal.

	Conversely, suppose
	\[
	\dim V=\dim W=n.
	\]
	Choose ordered bases $(v_1,\dots,v_n)$ of $V$ and $(w_1,\dots,w_n)$ of $W$. Define
	\[
	\left(\sum_{i=1}^n a_iv_i\right)^\tau
	=
	\sum_{i=1}^n a_iw_i.
	\]
	Proposition \ref{prop:coordinates} makes this definition unambiguous. The map preserves linear combinations and is bijective, with inverse obtained by interchanging the two bases.
\end{proof}

Dimension is therefore the complete invariant of a finite-dimensional vector space up to isomorphism. What depends on a basis is not the vector space itself, but its coordinate description.

\chapter{Linear Maps and Duality}

Chapter 1 described finite-dimensional vector spaces intrinsically, without using coordinates. We now study the maps which preserve their linear structure. The central theme remains the same: coordinates may be useful for calculation, but the objects and constructions themselves do not depend on a choice of basis.

\section{Linear maps}

\begin{definition}
Let $V$ and $W$ be vector spaces over the same field $F$. A map
\[
\tau\colon V\longrightarrow W,
\qquad v\longmapsto v^\tau,
\]
is \emph{linear} if
\[
(au+bv)^\tau=au^\tau+bv^\tau
\]
for all $u,v\in V$ and $a,b\in F$. The vector space of all linear maps from $V$ to $W$ will be denoted by $L(V,W)$.
\end{definition}

A linear map necessarily sends $0$ to $0$. The identity map on $V$ will be denoted by $I_V$, or simply $I$ when the underlying space is clear. The zero map will also be denoted by $0$.

We use right-action notation throughout. Thus, if
\[
V\xrightarrow{\tau}W\xrightarrow{\sigma}U,
\]
then
\[
v^{\tau\sigma}=(v^\tau)^\sigma.
\]
In particular, composition is read from left to right in the same order in which the maps act on a vector.

\begin{theorem}[Linear extension from a basis]\label{thm:linear-extension}
Let $B$ be a basis of $V$, and let $f\colon B\to W$ be any function. There exists a unique linear map
\[
\tau\colon V\to W
\]
such that $b^\tau=f(b)$ for every $b\in B$.
\end{theorem}

\begin{proof}
Every $v\in V$ has a unique expression as a finite linear combination
\[
v=a_1b_1+\cdots+a_nb_n,
\qquad b_i\in B.
\]
Define
\[
v^\tau=a_1f(b_1)+\cdots+a_nf(b_n).
\]
Uniqueness of the coordinates makes this well-defined, and the definition immediately gives linearity. Any linear map agreeing with $f$ on $B$ must satisfy the same formula, so it is unique.
\end{proof}

Thus a linear map is nothing more than an arbitrary assignment on a basis, followed by linear extension. In particular, two linear maps are equal if and only if they agree on one basis of their common domain.

\section{Kernel, image, and the first isomorphism theorem}

\begin{definition}
Let $\tau\in L(V,W)$. The \emph{kernel} and \emph{image} of $\tau$ are
\[
\ker\tau=\{v\in V\mid v^\tau=0\}
\]
and
\[
\im\tau=\{v^\tau\mid v\in V\}.
\]
\end{definition}

Both $\ker\tau$ and $\im\tau$ are subspaces of $V$ and $W$, respectively.

\begin{proposition}\label{prop:inj-surj}
A linear map $\tau\colon V\to W$ is injective if and only if
\[
\ker\tau=\{0\},
\]
and it is surjective if and only if
\[
\im\tau=W.
\]
\end{proposition}

\begin{proof}
The second statement is the definition of surjectivity. For the first, if $u^\tau=v^\tau$, then $(u-v)^\tau=0$. Hence $\tau$ is injective precisely when the only vector mapped to zero is zero itself.
\end{proof}

The kernel measures exactly how much information is lost by a linear map. Quotienting by the kernel removes that loss.

\begin{theorem}[First isomorphism theorem]\label{thm:first-isomorphism}
Let $\tau\colon V\to W$ be linear. Then there is a canonical isomorphism
\[
V/\ker\tau\cong\im\tau.
\]
Explicitly,
\[
(v+\ker\tau)^{\bar\tau}=v^\tau
\]
defines an isomorphism
\[
\bar\tau\colon V/\ker\tau\longrightarrow\im\tau.
\]
\end{theorem}

\begin{proof}
If $v+\ker\tau=u+\ker\tau$, then $v-u\in\ker\tau$, so
\[
v^\tau-u^\tau=(v-u)^\tau=0.
\]
Hence $\bar\tau$ is well-defined. It is linear by construction and surjective by the definition of $\im\tau$. If
\[
(v+\ker\tau)^{\bar\tau}=0,
\]
then $v\in\ker\tau$, so $v+\ker\tau$ is the zero coset. Thus $\bar\tau$ is injective.
\end{proof}

For finite-dimensional spaces, the familiar dimension formula is now a consequence rather than a separate counting argument.

\begin{definition}
For a linear map $\tau\colon V\to W$ with $V$ finite-dimensional, define
\[
\operatorname{rank}\tau=\dim\im\tau,
\qquad
\operatorname{nullity}\tau=\dim\ker\tau.
\]
\end{definition}

\begin{corollary}[Rank-nullity theorem]\label{cor:rank-nullity}
If $V$ is finite-dimensional and $\tau\colon V\to W$ is linear, then
\[
\dim V=\operatorname{nullity}\tau+\operatorname{rank}\tau.
\]
\end{corollary}

\begin{proof}
By Theorem \ref{thm:first-isomorphism},
\[
V/\ker\tau\cong\im\tau.
\]
Taking dimensions and using the quotient-dimension formula from Chapter 1 gives
\[
\dim V-\dim\ker\tau=\dim\im\tau.
\]
\end{proof}

\begin{corollary}\label{cor:endomorphism-injective-surjective}
Let $V$ be finite-dimensional and $\tau\in L(V,V)$. Then the following are equivalent:
\[
\tau\text{ is injective},\qquad
\tau\text{ is surjective},\qquad
\tau\text{ is an isomorphism}.
\]
\end{corollary}

\section{The vector space of linear maps}

The set $L(V,W)$ is itself a vector space. For $\tau,\sigma\in L(V,W)$ and $a,b\in F$, define
\[
v^{a\tau+b\sigma}=av^\tau+bv^\sigma.
\]
The vector-space axioms are inherited pointwise from $W$.

When $V=W$, we write
\[
\End(V)=L(V,V)
\]
and call its elements \emph{linear operators}. In addition to addition and scalar multiplication, $\End(V)$ has composition. Composition is associative, distributes over addition, and has identity $I_V$. We will exploit this richer structure later.

At this point no coordinates are needed. If ordered bases are eventually chosen, a linear map can of course be encoded by scalars. Such an encoding is a representation of the map relative to those choices, not the definition of the map itself.

\section{Dual spaces}

\begin{definition}
The \emph{dual space} of $V$ is
\[
V^*=L(V,F).
\]
Its elements are called \emph{linear functionals} on $V$.
\end{definition}

There is a canonical bilinear evaluation pairing
\[
V^*\times V\longrightarrow F,
\qquad
(\varphi,v)\longmapsto\varphi(v).
\]
Unlike coordinates, this pairing requires no choice of basis.

Suppose $V$ is finite-dimensional and $B=(v_1,\dots,v_n)$ is an ordered basis. By Theorem \ref{thm:linear-extension}, for each $i$ there is a unique functional $v^i\in V^*$ satisfying
\[
v^i(v_j)=\delta_{ij}.
\]
The ordered basis
\[
B^*=(v^1,\dots,v^n)
\]
is called the \emph{dual basis}.

\begin{proposition}\label{prop:dual-dimension}
If $V$ is finite-dimensional, then
\[
\dim V^*=\dim V.
\]
\end{proposition}

\begin{proof}
If $(v_1,\dots,v_n)$ is a basis of $V$, its dual basis $(v^1,\dots,v^n)$ is a basis of $V^*$. Indeed, for any $\varphi\in V^*$,
\[
\varphi=\sum_{i=1}^n\varphi(v_i)v^i,
\]
as can be checked on the basis vectors $v_j$.
\end{proof}

Every linear map induces a linear map in the opposite direction on dual spaces.

\begin{definition}
Let $\tau\colon V\to W$ be linear. Its \emph{dual map}
\[
\tau^*\colon W^*\longrightarrow V^*
\]
is defined by
\[
(\psi^{\tau^*})(v)=\psi(v^\tau).
\]
\end{definition}

The reversal of direction is essential. If
\[
V\xrightarrow{\tau}W\xrightarrow{\sigma}U,
\]
then
\[
(\tau\sigma)^*=\sigma^*\tau^*.
\]

There is also a canonical map from a finite-dimensional vector space to its double dual.

\begin{proposition}\label{prop:double-dual}
Define
\[
\iota_V\colon V\longrightarrow V^{**}
\]
by
\[
(v^{\iota_V})(\varphi)=\varphi(v).
\]
If $V$ is finite-dimensional, then $\iota_V$ is an isomorphism.
\end{proposition}

\begin{proof}
If $v^{\iota_V}=0$, then every functional vanishes on $v$. Extend $v$ to a basis when $v\neq0$ and take the corresponding dual functional which has value $1$ on $v$, a contradiction. Thus $\iota_V$ is injective. Since
\[
\dim V=\dim V^{**},
\]
it is an isomorphism.
\end{proof}

The importance of $\iota_V$ is that it is canonical: unlike an arbitrary isomorphism $V\cong V^*$, it requires no choice of basis.

\section{Tensor products}

The tensor product packages bilinear constructions into linear ones.

\begin{definition}
A \emph{tensor product} of vector spaces $V$ and $W$ is a vector space $V\otimes W$ together with a bilinear map
\[
\otimes\colon V\times W\longrightarrow V\otimes W,
\qquad
(v,w)\longmapsto v\otimes w,
\]
with the following universal property: for every vector space $U$ and every bilinear map
\[
b\colon V\times W\to U,
\]
there exists a unique linear map
\[
\widetilde b\colon V\otimes W\to U
\]
such that
\[
b(v,w)=(v\otimes w)^{\widetilde b}
\]
for all $v\in V$ and $w\in W$.
\end{definition}

A tensor product exists. One construction begins with the free vector space on the set $V\times W$ and quotients by the subspace generated by the relations
\[
(v+v',w)-(v,w)-(v',w),
\]
\[
(v,w+w')-(v,w)-(v,w'),
\]
and
\[
(av,w)-a(v,w),\qquad (v,aw)-a(v,w).
\]
The image of the symbol $(v,w)$ in the quotient is denoted by $v\otimes w$. The universal property shows that any two tensor products are canonically isomorphic, so the construction itself is secondary.

Every element of $V\otimes W$ is a finite sum of pure tensors $v\otimes w$, although such an expression is generally not unique.

\begin{proposition}\label{prop:tensor-basis}
If $(v_1,\dots,v_n)$ is a basis of $V$ and $(w_1,\dots,w_m)$ is a basis of $W$, then
\[
\{v_i\otimes w_j\mid 1\le i\le n,\ 1\le j\le m\}
\]
is a basis of $V\otimes W$. Consequently,
\[
\dim(V\otimes W)=(\dim V)(\dim W).
\]
\end{proposition}

\begin{proof}
Bilinearity shows that the displayed tensors span $V\otimes W$. Let $v^1,\dots,v^n$ and $w^1,\dots,w^m$ be the dual bases. For each pair $(r,s)$, the bilinear map
\[
(v,w)\longmapsto v^r(v)w^s(w)
\]
induces a linear functional on $V\otimes W$ which takes value $1$ on $v_r\otimes w_s$ and $0$ on all other displayed basis candidates. Hence they are linearly independent.
\end{proof}

Linear maps act naturally on tensor products. If
\[
\tau\colon V\to V',
\qquad
\sigma\colon W\to W',
\]
there is a unique linear map
\[
\tau\otimes\sigma\colon V\otimes W\to V'\otimes W'
\]
satisfying
\[
(v\otimes w)^{\tau\otimes\sigma}=v^\tau\otimes w^\sigma.
\]
This follows immediately from the universal property.

\section{Linear maps as tensors}

The dual space and tensor product combine to give a canonical description of linear maps.

For $\varphi\in V^*$ and $w\in W$, define a linear map
\[
\tau_{\varphi,w}\colon V\to W
\]
by
\[
v^{\tau_{\varphi,w}}=\varphi(v)w.
\]
The assignment $(\varphi,w)\mapsto\tau_{\varphi,w}$ is bilinear, hence induces a canonical linear map
\[
\Theta_{V,W}\colon V^*\otimes W\longrightarrow L(V,W)
\]
satisfying
\[
\Theta_{V,W}(\varphi\otimes w)=\tau_{\varphi,w}.
\]

\begin{theorem}\label{thm:hom-tensor}
If $V$ is finite-dimensional, then
\[
\Theta_{V,W}\colon V^*\otimes W\longrightarrow L(V,W)
\]
is an isomorphism.
\end{theorem}

\begin{proof}
Choose a basis $(v_1,\dots,v_n)$ of $V$ with dual basis $(v^1,\dots,v^n)$. For $\tau\in L(V,W)$, consider
\[
X_\tau=\sum_{i=1}^n v^i\otimes v_i^\tau.
\]
For $v=\sum_i a_iv_i$,
\[
v^{\Theta_{V,W}(X_\tau)}
=
\sum_i v^i(v)v_i^\tau
=
\sum_i a_iv_i^\tau
=
v^\tau.
\]
Thus $\Theta_{V,W}$ is surjective.

Every element of $V^*\otimes W$ can be written in the form
\[
X=\sum_{i=1}^n v^i\otimes w_i
\]
for some $w_i\in W$. If $\Theta_{V,W}(X)=0$, evaluation at $v_j$ gives $w_j=0$ for every $j$. Hence $X=0$, and $\Theta_{V,W}$ is injective.
\end{proof}

Although a basis was used in the proof, the map $\Theta_{V,W}$ itself was defined without one. The theorem therefore gives a basis-free identification
\[
\boxed{L(V,W)\cong V^*\otimes W.}
\]

\begin{corollary}\label{cor:hom-dimension}
If $V$ and $W$ are finite-dimensional, then
\[
\dim L(V,W)=(\dim V)(\dim W).
\]
\end{corollary}

In particular,
\[
\End(V)\cong V^*\otimes V.
\]
This is the form in which we will define trace.

\section{Contraction and trace}

The evaluation pairing gives a canonical linear map from $V^*\otimes V$ to the base field.

\begin{definition}
The \emph{contraction map}
\[
C_V\colon V^*\otimes V\longrightarrow F
\]
is the unique linear map satisfying
\[
C_V(\varphi\otimes v)=\varphi(v).
\]
\end{definition}

The existence and uniqueness of $C_V$ follow from the universal property of the tensor product, because evaluation $(\varphi,v)\mapsto\varphi(v)$ is bilinear.

\begin{definition}[Trace]\label{def:trace}
Let $V$ be finite-dimensional. Under the canonical isomorphism
\[
\Theta_{V,V}\colon V^*\otimes V\xrightarrow{\sim}\End(V),
\]
define
\[
\tr\colon\End(V)\longrightarrow F
\]
by
\[
\boxed{\tr=C_V\circ\Theta_{V,V}^{-1}.}
\]
\end{definition}

Thus trace is the contraction of the tensor corresponding to an endomorphism. No basis is part of the definition.

For a rank-one operator $\tau_{\varphi,v}$,
\[
\tr(\tau_{\varphi,v})=\varphi(v).
\]
Since every endomorphism is a sum of rank-one operators, this property determines trace completely.

\begin{proposition}\label{prop:trace-identity}
If $\dim V=n$, then
\[
\tr(I_V)=n\cdot1_F.
\]
\end{proposition}

\begin{proof}
Choose a basis $(v_1,\dots,v_n)$ with dual basis $(v^1,\dots,v^n)$. Under $\Theta_{V,V}$,
\[
I_V
\longleftrightarrow
\sum_{i=1}^n v^i\otimes v_i.
\]
Therefore
\[
\tr(I_V)
=
\sum_{i=1}^n v^i(v_i)
=
n\cdot1_F.
\]
\end{proof}

The fundamental property of trace is cyclicity.

\begin{theorem}[Cyclicity of trace]\label{thm:trace-cyclic}
For $\tau,\sigma\in\End(V)$,
\[
\tr(\tau\sigma)=\tr(\sigma\tau).
\]
\end{theorem}

\begin{proof}
By linearity, it is enough to prove the claim when $\tau$ has rank one. Write
\[
\tau=\tau_{\varphi,v},
\qquad
x^\tau=\varphi(x)v.
\]
Then
\[
x^{\tau\sigma}=\varphi(x)v^\sigma,
\]
so $\tau\sigma$ corresponds to $\varphi\otimes v^\sigma$, and hence
\[
\tr(\tau\sigma)=\varphi(v^\sigma).
\]
On the other hand,
\[
x^{\sigma\tau}=\varphi(x^\sigma)v=(\varphi^{\sigma^*})(x)v,
\]
so $\sigma\tau$ corresponds to $\varphi^{\sigma^*}\otimes v$, and
\[
\tr(\sigma\tau)
=
\varphi^{\sigma^*}(v)
=
\varphi(v^\sigma).
\]
Thus the two traces are equal.
\end{proof}

\begin{corollary}\label{cor:trace-conjugation}
If $\alpha\in\End(V)$ is invertible, then
\[
\tr(\alpha^{-1}\tau\alpha)=\tr(\tau)
\]
for every $\tau\in\End(V)$.
\end{corollary}

\begin{proof}
Using Theorem \ref{thm:trace-cyclic},
\[
\tr(\alpha^{-1}\tau\alpha)
=
\tr(\alpha\alpha^{-1}\tau)
=
\tr(\tau).
\]
\end{proof}

\begin{corollary}\label{cor:trace-commutator}
For $\tau,\sigma\in\End(V)$,
\[
\tr(\tau\sigma-\sigma\tau)=0.
\]
\end{corollary}

This last identity will reappear naturally when commutators are studied as Lie brackets.

The tensor description also explains the familiar coordinate formula without using it as a definition. If $(v_1,\dots,v_n)$ is a basis with dual basis $(v^1,\dots,v^n)$, then
\[
\Theta_{V,V}^{-1}(\tau)
=
\sum_{i=1}^n v^i\otimes v_i^\tau,
\]
and therefore
\[
\tr(\tau)=\sum_{i=1}^n v^i(v_i^\tau).
\]
Choosing coordinates turns this scalar into the sum of the diagonal coefficients of a representing matrix, but the scalar itself existed before any matrix was chosen.

We stop here. Determinants require exterior powers rather than merely tensor products, and the characteristic polynomial will arise from applying determinant to $tI-\tau$. Those constructions belong to the next chapter.

\section{Root system}
Suppose $E$ is an euclidean space, i.e., a finite dimensional vector space $V$ over $\R$ endowed with a positive definite symmetric bilinear form $(,)$. The \textbf{dual} of $\alpha\in E$ is defined as $\alpha^\vee = 2\alpha/(\alpha, \alpha)$. For any given vector $\alpha\in E$, we define a hyperplane $P_\alpha = \{\beta\in E \mid (\alpha, \beta) = 0\}$, which is the perpendicular complement of $\alpha$ with respect to $E$. The reflection with respect to $\alpha$ is the linear map acts trivially on $P_\alpha$ and sends $v$ to $-v$. Explicitly, $\sigma_\alpha(\beta) = \beta - (\beta, \alpha^\vee)\alpha$. Since $(\beta, \alpha^\vee)$ occurs frequently, we denote it by $\cartan{\beta}{\alpha}$. 



A subset $\Phi$ of an euclidean space $E$ is called a (reduced) root system on $E$ if it satisfied
\begin{enumerate}
	\item $\Phi$ is finite, excludes 0 and spans E.
	\item $\R\alpha \cap \Phi = {\pm \alpha}$ for every $\alpha \in \Phi$.
	\item $\sigma_{\alpha}(\beta) \in \Phi$ for every $\alpha, \beta \in \Phi$.
	\item $\cartan{\alpha}{\beta} \in \Z$.  
\end{enumerate}
And denote $(\Phi, E)$ as the root system with respect to $E$. Implicitly there is an underlying vector space $V$ corresponds to $E$. The dimension of $V$ is called the rank of $\Phi$.


Let $\alpha, \beta\in\Phi$, $\cartan{\alpha}{\beta}\cartan{\beta}{\alpha} = 4\cos^2\theta$ where $\theta$ is the angle between $\alpha$ and $\beta$ in $E$. Hence $4\cos^2\theta$ is limited to 0, 1, 2, 3, 4. The case 4 is trivial. For the case 1, 2 and 3, $\theta$ is given as $\pi/6(5\pi/6)$, $\pi/4(3\pi/4)$ and $\pi/3(2\pi/3)$ respectively. And as divisor of these numbers, at least one of $\cartan{\alpha}{\beta}$ or $\cartan{\beta}{\alpha}$ is $\pm1$. Due to this observation, we have the following 
\begin{proposition}\label {6.1}
Suppose $\alpha$ and $\beta$ are non-proportional roots, $(\alpha, \beta) < 0$ (resp. $(\alpha, \beta) > 0$), then $\alpha + \beta \in \Phi$ (resp. $\alpha - \beta \in \Phi$).
\end{proposition}

\begin{proof}
	 Suppose $(\alpha, \beta) < 0$ and $\cartan{\alpha}{\beta} = -1$, $\alpha + \beta = \sigma_\beta(\alpha) \in \Phi$; similarly for $\cartan{\beta}{\alpha}$. Similarly for $(\alpha, \beta) > 0$.
\end{proof}

As an application, let $\alpha$ and $\beta$ be non-proportional roots. Consider the $\beta$-string through $\alpha$, i.e., the set $S = \{\alpha + i\beta\mid i\in \Z\}$, which is invariant under $\sigma_\beta$, hence $S\cap\Phi$ is also invariant under $\sigma_\beta$. Let $q,\ r\in\Z$ be maximal that $\alpha + q\beta, \ \alpha - r\beta \in \Phi$, we must have $\sigma_\beta(\alpha - r\beta) = \alpha - \cartan{\alpha}{\beta}\beta + r\beta = \alpha + q\beta$, $\cartan{\alpha}{\beta} = r - q \le 3$, so the length of the string is at most 4. We next show the string $S\cap\Phi$ is unbroken. Suppose otherwise, there is $i\in \Z$ and $-r < i < q$ such that the string is broken at $\alpha + i\beta$, we have $(\alpha + i_1\beta, \beta) \ge 0$ and $(\alpha + i_2\beta, \beta) \le 0$ for some $i_1 < i < i_2$ due to proposition \ref{6.1}, which is absurd.

A root system $\Phi$ is \textbf{reducible} if it can be partitioned to nonempty orthogonal components, otherwise it is \textbf{irreducible}.  


\section{Weyl group}
The following fact is useful,

\begin{lemma}{\label{sdf}}
	Let $\Phi$ be a finite set spans $E$, an euclidean space on $V$, in which all the reflections $\sigma_\alpha$ permute $\Phi$. Let $\sigma \in GL(V)$ acts trivially on a hyperplane $P$ in $E$ and sends some $\alpha\in E$ to its negative. Then $\sigma = \sigma_\alpha$.
\end{lemma}
\begin{proof}
	Let $\tau = \sigma\sigma_\alpha$. Both $\sigma$ and $\sigma_\alpha$ acts as identity on $E/\R\alpha$, so does $\tau$. Hence the minimal polynomial $m(T)$ of $\tau$ on $E$ divides $(T - 1)^{(\dim E)}$. Since $\tau$ permutes $\Phi$, $\tau^{n!}$ acts trivially on $\Phi$, where $n = |\Phi|$, it follows that $\tau^{n!} = 1$, i.e., $m(T)$ divides $\text{g.c.d}((T^{n!} - 1), (T - 1)^{(\dim E)}) = T - 1$, $\tau = 1$.
\end{proof}


Let $(\Phi, E)$ be a root system, the dual of $\Phi$ is the set of all the dual of elements in $\Phi$, denoted by $\Phi^\vee$. It is a root system in $E$. Call root system $(\Phi, E)$ and $(\Phi', E')$ \textbf{isomorphic} if there exists a vector space isomorphism $\tau \colon V \to V'$ sending $\Phi \to \Phi'$ and  $\cartan{\alpha}{\beta} = \cartan{\tau(\alpha)}{\tau(\beta)}$ for each pair of roots $\alpha, \beta \in \Phi$. $\tau$ preserves $\langle \ ,\  \rangle$ while is not necessary a isometry. It follows at once that $\tau(\sigma_\alpha(\beta)) = \sigma_{\tau(\alpha)}(\tau(\beta))$. When there is no ambiguity, we denote $\Aut(\Phi)$ the group of all automorphisms of $(\Phi, E)$. Each reflection $\sigma_\alpha(\alpha\in\Phi)$ is an automorphism as well as isometry. The group $\W(\Phi)$ generated by all the reflections $\sigma_\alpha$ is called \textbf{Weyl group} of the $\Phi$. There is a canonical isomorphism between $\W(\Phi)$ and $\W(\Phi^\vee)$ even though $\Phi$ is sometimes not isomorphic to $\Phi'$.

Suppose $\sigma \in GL(V)$ permutes $\Phi$. Then $\sigma \sigma_\alpha \sigma^{-1}$ acts on $\sigma(\beta)$ as $\sigma \sigma_\alpha \sigma^{-1} \sigma(\beta)$ = $\sigma(\sigma_\alpha(\beta)) = \sigma(\beta) - \cartan{\alpha}{\beta}\sigma(\alpha)$. So $\sigma \sigma_\alpha \sigma^{-1}$ acts trivially on $\sigma(P_\alpha)$ and sends $\sigma(\alpha)$ to its negative, which is exactly the reflection $\sigma_{\sigma(\alpha)}$ by lemma \ref{sdf}, from which we conclude that $\W(\Phi) \triangleleft \Aut(\Phi)$. Moreover, $\sigma_{\sigma(\alpha)} \sigma(\beta)= \sigma(\beta) - \cartan{\sigma(\alpha)}{\sigma(\beta)}\sigma(\alpha)$, it follows that $\cartan{\alpha}{\beta} = \cartan{\sigma(\alpha)}{\sigma(\beta)}$, $\sigma\in\Aut(\Phi)$.
